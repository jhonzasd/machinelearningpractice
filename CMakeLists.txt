cmake_minimum_required(VERSION 3.16)
project(Practice2Regression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(regression src/regression.cpp src/dataset.cpp)
target_include_directories(regression PUBLIC include)
if(MSVC)
  target_compile_options(regression PRIVATE /W4 /utf-8)
else()
  target_compile_options(regression PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(regression_demo src/main.cpp)
target_link_libraries(regression_demo PRIVATE regression)
add_executable(regression_tests tests/test_regression.cpp)
target_link_libraries(regression_tests PRIVATE regression)
enable_testing()
add_test(NAME regression_suite COMMAND regression_tests ${CMAKE_CURRENT_SOURCE_DIR}/data)
